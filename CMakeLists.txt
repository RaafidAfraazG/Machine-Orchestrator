cmake_minimum_required(VERSION 3.16)
project(MachineOrchestrator LANGUAGES CXX)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_ASAN "Address sanitizer" OFF)
option(ENABLE_UBSAN "Undefined behavior sanitizer" OFF)
option(ENABLE_TSAN "Thread sanitizer" OFF)
if(ENABLE_ASAN AND ENABLE_TSAN)
  message(FATAL_ERROR "ASAN and TSAN are mutually exclusive")
endif()
add_compile_options(-Wall -Wextra -Wpedantic)
foreach(pair ASAN UBSAN TSAN)
  if(ENABLE_${pair})
    if(pair STREQUAL "ASAN")
      set(san address)
    elseif(pair STREQUAL "UBSAN")
      set(san undefined)
    else()
      set(san thread)
    endif()
    add_compile_options(-fsanitize=${san} -fno-omit-frame-pointer)
    add_link_options(-fsanitize=${san})
  endif()
endforeach()
find_package(Threads REQUIRED)
add_library(mo src/protocol.cpp src/transport.cpp src/reactor.cpp src/device.cpp src/event_log.cpp src/resource_manager.cpp src/scheduler.cpp
  src/deadlock_detector.cpp src/watchdog.cpp src/engine.cpp)
set(MO_INJECT_BUG "" CACHE STRING "Deliberate mutation; NEVER use in production")
if(NOT MO_INJECT_BUG STREQUAL "")
  set(bug_source "")
  if(MO_INJECT_BUG STREQUAL "unlocked_log")
    set(bug_source "src/event_log.cpp")
  endif()
  if(MO_INJECT_BUG STREQUAL "unlocked_queue")
    set(bug_source "src/scheduler.cpp")
  endif()
  if(MO_INJECT_BUG STREQUAL "lock_order")
    set(bug_source "src/resource_manager.cpp")
  endif()
  if(MO_INJECT_BUG STREQUAL "missing_abort_release")
    set(bug_source "src/engine.cpp")
  endif()
  if(MO_INJECT_BUG STREQUAL "lost_wakeup")
    set(bug_source "src/resource_manager.cpp")
  endif()
  if(MO_INJECT_BUG STREQUAL "wrong_priority")
    set(bug_source "src/scheduler.cpp")
  endif()
  if(MO_INJECT_BUG STREQUAL "aging_disabled")
    set(bug_source "src/scheduler.cpp")
  endif()
  if(MO_INJECT_BUG STREQUAL "missing_cycle_edge")
    set(bug_source "src/deadlock_detector.cpp")
  endif()
  if(MO_INJECT_BUG STREQUAL "watchdog_disabled")
    set(bug_source "src/watchdog.cpp")
  endif()
  if(MO_INJECT_BUG STREQUAL "timeout_off_by_one")
    set(bug_source "src/device.cpp")
  endif()
  if(MO_INJECT_BUG STREQUAL "transaction_mismatch")
    set(bug_source "src/device.cpp")
  endif()
  if(MO_INJECT_BUG STREQUAL "crc_skipped")
    set(bug_source "src/protocol.cpp")
  endif()
  if(MO_INJECT_BUG STREQUAL "double_release")
    set(bug_source "src/resource_manager.cpp")
  endif()
  if(MO_INJECT_BUG STREQUAL "callback_after_complete")
    set(bug_source "src/device.cpp")
  endif()
  if(MO_INJECT_BUG STREQUAL "missing_release_event")
    set(bug_source "src/resource_manager.cpp")
  endif()
  if(MO_INJECT_BUG STREQUAL "drop_counter")
    set(bug_source "src/event_log.cpp")
  endif()
  if(bug_source STREQUAL "")
    message(FATAL_ERROR "Unknown MO_INJECT_BUG")
  endif()
  set_source_files_properties(${bug_source} PROPERTIES COMPILE_DEFINITIONS "MO_INJECT_BUG=\"${MO_INJECT_BUG}\"")
endif()
target_include_directories(mo PUBLIC include)
target_link_libraries(mo PUBLIC Threads::Threads)
add_executable(mo_demo src/main_demo.cpp)
target_link_libraries(mo_demo PRIVATE mo)
add_executable(mo_orchestrator_demo src/main_orchestrator_demo.cpp)
target_link_libraries(mo_orchestrator_demo PRIVATE mo)
add_executable(mo_bench_latency src/bench_latency.cpp)
target_link_libraries(mo_bench_latency PRIVATE mo)
include(CTest)

  include(FetchContent)
  FetchContent_Declare(googletest
    URL https://codeload.github.com/google/googletest/tar.gz/refs/tags/v1.15.2
    URL_HASH SHA256=7b42b4d6ed48810c5362c265a17faebe90dc2373c885e5216439d37927f02926)
