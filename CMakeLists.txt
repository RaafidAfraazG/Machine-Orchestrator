cmake_minimum_required(VERSION 3.16)
project(MachineOrchestrator LANGUAGES CXX)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_ASAN "Address sanitizer" OFF)
option(ENABLE_UBSAN "Undefined behavior sanitizer" OFF)
option(ENABLE_TSAN "Thread sanitizer" OFF)
if(ENABLE_ASAN AND ENABLE_TSAN)
  message(FATAL_ERROR "ASAN and TSAN are mutually exclusive")
endif()
add_compile_options(-Wall -Wextra -Wpedantic)
foreach(pair ASAN UBSAN TSAN)
  if(ENABLE_${pair})
    if(pair STREQUAL "ASAN")
      set(san address)
    elseif(pair STREQUAL "UBSAN")
      set(san undefined)
    else()
      set(san thread)
    endif()
    add_compile_options(-fsanitize=${san} -fno-omit-frame-pointer)
    add_link_options(-fsanitize=${san})
  endif()
endforeach()
find_package(Threads REQUIRED)
add_library(mo src/protocol.cpp src/transport.cpp src/reactor.cpp src/device.cpp)
target_include_directories(mo PUBLIC include)
target_link_libraries(mo PUBLIC Threads::Threads)
add_executable(mo_demo src/main_demo.cpp)
target_link_libraries(mo_demo PRIVATE mo)
include(CTest)
if(BUILD_TESTING)
  include(FetchContent)
  FetchContent_Declare(googletest URL https://codeload.github.com/google/googletest/tar.gz/refs/tags/v1.15.2)
  FetchContent_MakeAvailable(googletest)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(mo_tests tests/tests.cpp)
  target_link_libraries(mo_tests PRIVATE mo GTest::gtest_main util)
  target_compile_definitions(mo_tests PRIVATE MO_SIM="${CMAKE_CURRENT_SOURCE_DIR}/sim/device_sim.py" MO_PYTHON="${Python3_EXECUTABLE}")
  add_test(NAME foundation COMMAND mo_tests)
  set_tests_properties(foundation PROPERTIES TIMEOUT 60)
endif()
